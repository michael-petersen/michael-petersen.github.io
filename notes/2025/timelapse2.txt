pattern = "output_{:04d}.jpg"

def load_image(path):
    return np.array(Image.open(path).convert("RGB"), dtype=np.uint8)

# Load first image to establish the stack
first_path = pattern.format(start)
result = load_image(first_path)

for i in range(start + 1, end + 1):
    path = pattern.format(i)
    img = load_image(path)
    if img.shape != result.shape:
        raise ValueError(f"{path} has shape {img.shape}, expected {result.shape}")
